cmake_minimum_required(VERSION 3.24)
project(BackwardModuleStrings LANGUAGES CXX)
if(NOT MSVC)
    message(FATAL_ERROR "This regression exercises the Windows DbgHelp adapter with MSVC AddressSanitizer")
endif()
add_executable(BackwardModuleStrings Main.cpp)
target_include_directories(BackwardModuleStrings PRIVATE ../../../../../external/include)
target_compile_options(BackwardModuleStrings PRIVATE /fsanitize=address /Zi)
target_link_options(BackwardModuleStrings PRIVATE /INCREMENTAL:NO /DEBUG)
target_link_libraries(BackwardModuleStrings PRIVATE dbghelp psapi)
enable_testing()
foreach(CASE strings path-failure name-failure module-failure resolver-module-failure symbol-failure line-size)
    add_test(NAME ${CASE} COMMAND BackwardModuleStrings ${CASE})
endforeach()
